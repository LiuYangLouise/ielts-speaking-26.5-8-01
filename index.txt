            fillTopicList('part1-keep-topics', part1Keep, 'keep');
            fillTopicList('part1-new-topics', part1New, 'new');
            
            const p2KeepAll = [...peopleKeep, ...eventsKeep, ...objectsKeep, ...placesKeep];
            fillTopicList('p2-keep-topics', p2KeepAll, 'keep');
            fillTopicList('part2-new-topics', part2New, 'new');
            
            setupEventListeners();
            
            // 默认选中第一个话题
            setTimeout(() => {
                const firstTopic = document.querySelector('#old-part1-topics .topic-item');
                if (firstTopic) {
                    firstTopic.classList.add('active');
                    const topicId = firstTopic.getAttribute('data-id');
                    findAndShowTopic(topicId);
                }
            }, 100);
        }

        function fillTopicList(listId, topics, type) {
            const list = document.getElementById(listId);
            if (!list) return;
            topics.forEach(topic => {
                const item = document.createElement('div');
                item.className = `topic-item ${type}`;
                item.textContent = topic.title;
                item.setAttribute('data-id', topic.id);
                list.appendChild(item);
            });
        }

        function setupEventListeners() {
            document.querySelectorAll('.category-btn').forEach(btn => {
                btn.addEventListener('click', function(e) {
                    e.stopPropagation();
                    const category = this.getAttribute('data-category');
                    const targetList = document.getElementById(`${category}-topics`);
                    const icon = this.querySelector('.fa-chevron-down, .fa-chevron-up');
                    if (targetList.classList.contains('active')) {
                        targetList.classList.remove('active');
                        if (icon) icon.className = 'fas fa-chevron-down icon';
                    } else {
                        targetList.classList.add('active');
                        if (icon) icon.className = 'fas fa-chevron-up icon';
                    }
                });
            });

            document.querySelectorAll('.topic-item').forEach(item => {
                item.addEventListener('click', function() {
                    document.querySelectorAll('.topic-item').forEach(t => t.classList.remove('active'));
                    this.classList.add('active');
                    const topicId = this.getAttribute('data-id');
                    findAndShowTopic(topicId);
                });
            });

            document.querySelectorAll('.part-btn').forEach(btn => {
                btn.addEventListener('click', function() {
                    const part = this.getAttribute('data-part');
                    document.querySelectorAll('.part-btn').forEach(b => b.classList.remove('active'));
                    this.classList.add('active');
                    currentPart = part;
                    if (currentTopic) {
                        if (part === 'part1' && currentTopic.part1) showPart1(currentTopic);
                        else if (part === 'part2' && currentTopic.part2) showPart2(currentTopic);
                        else if (part === 'part3' && currentTopic.part3) showPart3(currentTopic);
                    }
                    resetTimerForPart();
                });
            });

            document.getElementById('start-timer').addEventListener('click', toggleTimer);
            document.getElementById('reset-timer').addEventListener('click', resetTimer);
            
            document.getElementById('searchInput').addEventListener('input', function(e) {
                const keyword = e.target.value.toLowerCase().trim();
                document.querySelectorAll('.topic-item').forEach(item => {
                    const title = item.textContent.toLowerCase();
                    if (keyword === '' || title.includes(keyword)) {
                        item.style.display = 'block';
                        const parentList = item.closest('.topic-list');
                        if (parentList && keyword !== '') {
                            parentList.classList.add('active');
                            const btn = parentList.previousElementSibling;
                            if (btn && btn.classList.contains('category-btn')) {
                                const icon = btn.querySelector('.fa-chevron-down, .fa-chevron-up');
                                if (icon) icon.className = 'fas fa-chevron-up icon';
                            }
                        }
                    } else {
                        item.style.display = 'none';
                    }
                });
            });
        }

        function findAndShowTopic(topicId) {
            const topic = allTopics.find(t => t.id === topicId);
            if (topic) {
                currentTopic = topic;
                const cardHeader = document.getElementById('card-header');
                const topicBadge = document.getElementById('topic-badge');
                cardHeader.className = 'card-header';
                if (topic.batch === 'new') {
                    cardHeader.classList.add('new');
                    topicBadge.className = 'badge-new';
                    topicBadge.textContent = '新增题';
                } else if (topic.batch === 'keep') {
                    cardHeader.classList.add('keep');
                    topicBadge.className = 'badge-keep';
                    topicBadge.textContent = '保留题';
                } else {
                    cardHeader.classList.add('old');
                    topicBadge.className = 'badge-old';
                    topicBadge.textContent = '万年老题';
                }
                
                if (currentPart === 'part1' && topic.part1) showPart1(topic);
                else if (currentPart === 'part2' && topic.part2) showPart2(topic);
                else if (currentPart === 'part3' && topic.part3) showPart3(topic);
                else {
                    if (topic.part1) document.querySelector('.part-btn[data-part="part1"]').click();
                    else if (topic.part2) document.querySelector('.part-btn[data-part="part2"]').click();
                    else if (topic.part3) document.querySelector('.part-btn[data-part="part3"]').click();
                }
                resetTimerForPart();
            }
        }

        function showPart1(topic) {
            const content = document.getElementById('card-content');
            content.innerHTML = `<div class="topic-title ${topic.batch}">${topic.title}</div><div class="task-instruction ${topic.batch}">Answer the following questions:</div><div class="question-list ${topic.batch}">${topic.part1.questions.map((q, i) => `<div class="question-item ${topic.batch}"><strong>${i+1}.</strong> ${q}</div>`).join('')}</div>`;
        }

        function showPart2(topic) {
            const content = document.getElementById('card-content');
            content.innerHTML = `<div class="topic-title ${topic.batch}">${topic.title}</div><div class="task-instruction ${topic.batch}">${topic.part2.instruction}</div><div class="card-item ${topic.batch}"><h3>You should say:</h3><ul>${topic.part2.youShouldSay.map(item => `<li>${item}</li>`).join('')}</ul></div>`;
        }

        function showPart3(topic) {
            const content = document.getElementById('card-content');
            content.innerHTML = `<div class="topic-title ${topic.batch}">${topic.title}</div><div class="task-instruction ${topic.batch}">Answer the follow-up questions:</div><div class="part3-section ${topic.batch}"><ul>${topic.part3.questions.map((q, i) => `<li><strong>${i+1}.</strong> ${q}</li>`).join('')}</ul></div>`;
        }

        function resetTimerForPart() {
            clearInterval(timerInterval);
            isTimerRunning = false;
            if (currentPart === 'part2') {
                timerSeconds = 60;
                document.getElementById('timer').textContent = '01:00';
                document.getElementById('timer-label').textContent = '准备时间';
                isPart2Mode = true;
            } else {
                timerSeconds = 0;
                document.getElementById('timer').textContent = '00:00';
                document.getElementById('timer-label').textContent = '答题时间';
                isPart2Mode = false;
            }
            document.getElementById('start-timer').innerHTML = '<i class="fas fa-play-circle"></i> 开始';
        }

        function resetTimer() {
            clearInterval(timerInterval);
            isTimerRunning = false;
            resetTimerForPart();
        }

        function toggleTimer() {
            if (isTimerRunning) {
                clearInterval(timerInterval);
                isTimerRunning = false;
                document.getElementById('start-timer').innerHTML = '<i class="fas fa-play-circle"></i> 开始';
            } else {
                isTimerRunning = true;
                document.getElementById('start-timer').innerHTML = '<i class="fas fa-pause-circle"></i> 暂停';
                timerInterval = setInterval(() => {
                    if (currentPart === 'part2' && isPart2Mode) {
                        if (timerSeconds > 0) timerSeconds--;
                        if (timerSeconds === 0) {
                            isPart2Mode = false;
                            document.getElementById('timer-label').textContent = '答题时间';
                        }
                    } else {
                        timerSeconds++;
                        if (currentPart === 'part2' && timerSeconds === 120) {
                            const timerDisplay = document.getElementById('timer');
                            timerDisplay.style.color = '#ff6b6b';
                            setTimeout(() => { timerDisplay.style.color = ''; }, 500);
                        }
                    }
                    const mins = Math.floor(timerSeconds / 60);
                    const secs = timerSeconds % 60;
                    document.getElementById('timer').textContent = `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
                }, 1000);
            }
        }

        document.addEventListener('DOMContentLoaded', buildSidebar);
    </script>
